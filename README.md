        let currentQueue = [];
        let currentPartIndex = 0;
        let history = [];
        let isFirstTry = true;

        function initDashboard() {
            const grid = document.getElementById('part-grid');
            grid.innerHTML = '';
            for (let i = 0; i < 7; i++) {
                const btn = document.createElement('div');
                btn.className = "bg-slate-800 p-6 rounded-2xl border border-slate-700 hover:border-blue-500 cursor-pointer transition-all hover:-translate-y-1 group";
                btn.innerHTML = `
                    <div class="text-blue-500 font-bold text-xs uppercase mb-1">Phần ${i + 1}</div>
                    <div class="text-lg font-bold group-hover:text-blue-400">Câu ${i * 50 + 1} - ${(i + 1) * 50}</div>
                    <p class="text-slate-500 text-sm mt-2">Nhấn để bắt đầu ôn tập phần này.</p>
                `;
                btn.onclick = () => startPart(i);
                grid.appendChild(btn);
            }
        }

        function shuffle(array) {
            return array.sort(() => Math.random() - 0.5);
        }

        function startPart(idx) {
            currentPartIndex = idx;
            const start = idx * 50;
            const partQuestions = allQuestions.slice(start, start + 50);
            currentQueue = shuffle([...partQuestions]);
            history = [];
            
            document.getElementById('dashboard').classList.add('hidden');
            document.getElementById('summary-screen').classList.add('hidden');
            document.getElementById('quiz-screen').classList.remove('hidden');
            document.getElementById('part-title').innerText = `Phần ${idx + 1}`;
            
            showQuestion();
        }

        function showQuestion() {
            if (currentQueue.length === 0) {
                showSummary();
                return;
            }

            isFirstTry = true;
            const q = currentQueue[0];
            document.getElementById('question-text').innerText = q.q;
            document.getElementById('progress-text').innerText = `${50 - currentQueue.length}/50`;
            document.getElementById('feedback').classList.add('hidden');
            document.getElementById('next-btn').classList.add('hidden');

            const grid = document.getElementById('options-grid');
            grid.innerHTML = '';
            
            // Randomize options display
            const opts = q.a.map((text, index) => ({ text, index }));
            shuffle(opts).forEach(opt => {
                const b = document.createElement('button');
                b.className = "option-btn w-full text-left p-4 rounded-xl bg-slate-700/30 font-medium";
                b.innerText = opt.text;
                b.onclick = () => handleAnswer(b, opt.index === q.c, q);
                grid.appendChild(b);
            });
        }

        function handleAnswer(btn, isCorrect, qObj) {
            const feedback = document.getElementById('feedback');
            const nextBtn = document.getElementById('next-btn');
            
            if (isCorrect) {
                btn.classList.add('correct');
                if (isFirstTry) {
                    feedback.className = "mt-8 p-4 rounded-xl border border-emerald-500/50 bg-emerald-500/10 text-emerald-400";
                    feedback.innerHTML = "<b>Chính xác!</b> Bạn đã thuộc câu này.";
                    history.push({ ...qObj, status: 'correct', userAns: btn.innerText });
                    currentQueue.shift();
                } else {
                    feedback.className = "mt-8 p-4 rounded-xl border border-yellow-500/50 bg-yellow-500/10 text-yellow-400";
                    feedback.innerHTML = "<b>Đúng rồi!</b> Nhưng vì bạn đã sai lần đầu, câu này sẽ quay lại sau.";
                    const temp = currentQueue.shift();
                    currentQueue.push(temp);
                    shuffle(currentQueue);
                }
            } else {
                if (isFirstTry) {
                    history.push({ ...qObj, status: 'wrong', userAns: btn.innerText });
                }
                isFirstTry = false;
                btn.classList.add('wrong');
                feedback.className = "mt-8 p-4 rounded-xl border border-red-500/50 bg-red-500/10 text-red-400";
                feedback.innerHTML = `<b>Sai rồi!</b> Đáp án đúng là: <br><span class="text-white font-semibold">${qObj.a[qObj.c]}</span>`;
                
                // Show correct answer
                Array.from(document.getElementById('options-grid').children).forEach(b => {
                    if (b.innerText === qObj.a[qObj.c]) b.classList.add('correct');
                });
            }
            
            feedback.classList.remove('hidden');
            nextBtn.classList.remove('hidden');
            // Disable other buttons
            Array.from(document.getElementById('options-grid').children).forEach(b => b.disabled = true);
        }

        function showSummary() {
            document.getElementById('quiz-screen').classList.add('hidden');
            document.getElementById('summary-screen').classList.remove('hidden');
            document.getElementById('review-list').classList.add('hidden');
        }

        function showReview() {
            const list = document.getElementById('review-list');
            list.innerHTML = '<h3 class="text-xl font-bold mb-4 border-b border-slate-700 pb-2">Chi tiết bài làm:</h3>';
            history.forEach((item, i) => {
                const div = document.createElement('div');
                div.className = `p-4 rounded-xl border ${item.status === 'correct' ? 'border-emerald-500/20 bg-emerald-500/5' : 'border-red-500/20 bg-red-500/5'}`;
                div.innerHTML = `
                    <p class="font-bold mb-2 text-slate-200">${i+1}. ${item.q}</p>
                    <p class="text-sm text-slate-400">Đáp án của bạn: <span class="${item.status === 'correct' ? 'text-emerald-400' : 'text-red-400'}">${item.userAns}</span></p>
                    ${item.status === 'wrong' ? `<p class="text-sm text-emerald-400">Đáp án đúng: ${item.a[item.c]}</p>` : ''}
                `;
                list.appendChild(div);
            });
            list.classList.remove('hidden');
        }

        function goHome() {
            document.getElementById('quiz-screen').classList.add('hidden');
            document.getElementById('summary-screen').classList.add('hidden');
            document.getElementById('dashboard').classList.remove('hidden');
            initDashboard();
        }

        document.getElementById('next-btn').onclick = showQuestion;
        window.onload = initDashboard;
